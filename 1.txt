    PROCEDURE QteStock@1000000016(No@1000000000 : Code[20];D@1000000001 : Date) : Decimal;
    BEGIN
      // quantite en stock a la date = somme des ecritures articles (table 32)
      CLEAR(ILEs);
      ILEs.SETCURRENTKEY("Item No.","Posting Date");
      ILEs.SETRANGE("Item No.",No);
      ILEs.SETRANGE("Posting Date",0D,D);
      ILEs.CALCSUMS(Quantity);
      EXIT(ILEs.Quantity);
    END;

    PROCEDURE FDate@1000000010(D@1000000000 : Date) : Text[10];
    BEGIN
      IF D = 0D THEN
        EXIT('');
      EXIT(FORMAT(D,0,'<Year4>-<Month,2>-<Day,2>'));
    END;

    PROCEDURE T@1000000011(Txt@1000000000 : Text[250]) : Text[250];
    BEGIN
      // retire les separateurs et guillemets
      EXIT(DELCHR(CONVERTSTR(Txt,';',','),'=','"'));
    END;

    PROCEDURE N@1000000012(Val@1000000000 : Decimal) : Text[30];
    BEGIN
      // nombre au format XML (point decimal, sans separateur de milliers)
      EXIT(FORMAT(Val,0,9));
    END;

    PROCEDURE AgenceDe@1000000014(Charge@1000000000 : Code[20]) : Text[20];
    BEGIN
      IF Charge = '' THEN
        EXIT('');
      IF SP.GET(COPYSTR(Charge,1,MAXSTRLEN(SP.Code))) THEN
        EXIT(FORMAT(SP."Sales Resp. Ctr. Filter"));
      EXIT('');
    END;

    PROCEDURE CoutDe@1000000015(No@1000000000 : Code[20];D@1000000001 : Date;Qte@1000000002 : Decimal) : Decimal;
    BEGIN
      // cout = quantite x cout unitaire table 50007 en vigueur a la FIN DU MOIS de la vente
      // MAJ 22/10/26 : SrcCout = 50007 / FICHE / AUCUN ; cout fiche article si option CoutSecours
      SrcCout := '';
      D := CALCDATE('<CM>',D);
      IF (No = '') OR (Qte = 0) THEN
        EXIT(0);
      CLEAR(TarifC);
      TarifC.SETCURRENTKEY("Item No.","Starting Date");
      TarifC.SETRANGE("Item No.",No);
      TarifC.SETFILTER("Starting Date",'<=%1',D);
      IF TarifC.FINDLAST THEN BEGIN
        SrcCout := '50007';
        EXIT(ROUND(Qte * TarifC."Unit Cost",0.001));
      END;
      SrcCout := 'AUCUN';
      IF CoutSecours THEN
        IF ItemC.GET(No) THEN
          IF ItemC."Unit Cost" <> 0 THEN BEGIN
            SrcCout := 'FICHE';
            EXIT(ROUND(Qte * ItemC."Unit Cost",0.001));
          END;
      EXIT(0);
    END;

    PROCEDURE FamItem@1000000040(No@1000000000 : Code[20]);
    BEGIN
      IF No <> '' THEN
        IF ItemF.GET(No) THEN
          IF ItemF."Grande Famille Budg‚taire" <> '' THEN BEGIN
            GF := ItemF."Grande Famille Budg‚taire";
            FB := ItemF."Famille Budg‚taire";
            SFB := ItemF."Sous Famille Budg‚taire";
          END;
      GFL := '';
      FBL := '';
      SFBL := '';
      ItemCat.RESET;
      ItemCat.SETRANGE(Budgetaire,TRUE);
      ItemCat.SETRANGE(Code,GF);
      IF ItemCat.FINDFIRST THEN
        GFL := ItemCat.Description;
      ProdGrp.RESET;
      ProdGrp.SETRANGE(Budgetaire,TRUE);
      ProdGrp.SETRANGE("Item Category Code",GF);
      ProdGrp.SETRANGE(Code,FB);
      IF ProdGrp.FINDFIRST THEN
        FBL := ProdGrp.Description;
      Manuf.RESET;
      Manuf.SETRANGE(Budgetaire,TRUE);
      Manuf.SETRANGE("Item Category Code",GF);
      Manuf.SETRANGE("Product Group Code",FB);
      Manuf.SETRANGE(Code,SFB);
      IF Manuf.FINDFIRST THEN
        SFBL := Manuf.Name;
    END;

    PROCEDURE LocalDe@1000000041(RC@1000000000 : Code[20];Usr@1000000001 : Text[50]) : Text[20];
    BEGIN
      IF RC <> '' THEN
        EXIT(Agence(RC));
      IF Usr <> '' THEN
        IF UserS.GET(COPYSTR(Usr,1,MAXSTRLEN(UserS."User ID"))) THEN
          IF UserS."Sales Resp. Ctr. Filter" <> '' THEN
            EXIT(Agence(COPYSTR(UserS."Sales Resp. Ctr. Filter",1,20)));
      EXIT(Agence(''));
    END;

    PROCEDURE Agence@1000000013(Code@1000000000 : Code[20]) : Text[20];
    BEGIN
      IF Code = '' THEN
        EXIT('TUNIS');
      IF Code = 'CHR QNQ' THEN
        EXIT('CHR PLACO');
      EXIT(Code);
    END;

    BEGIN
    END.
