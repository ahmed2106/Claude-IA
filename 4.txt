    { [{D896C78B-AAF3-4718-8F59-5B80A8AA3353}];  ;Root                ;Element ;Text     }

    { [{888F7DCA-7E54-4289-BC60-F4987BB0C9A5}];1 ;ENTETE              ;Element ;Table   ;
                                                  SourceTable=Table2000000026;
                                                  SourceTableView=WHERE(Field1=CONST(1));
                                                  Export::OnPreXMLItem=BEGIN
                                                                         IF DateFin = 0D THEN
                                                                           ERROR('Renseignez la date d arrete.');
                                                                         // V2 : une date d arrete future gonfle les retards et inclut les ecritures post-datees
                                                                         IF DateFin > TODAY THEN
                                                                           DateFin := TODAY;
                                                                       END;
                                                                        }

    { [{34824DB3-F42A-444F-8403-2EE880E7D8E7}];2 ;LigneEntete         ;Element ;Text    ;
                                                  VariableName=LigneEntete;
                                                  Export::OnBeforePassVariable=BEGIN
                                                                                 LigneEntete :=
                                                                                   'Source;DateArrete;Fournisseur;NomFournisseur;Acheteur;GroupeFournisseur;Document;TypeDoc;DateComptable;DateEcheance;' +
                                                                                   'MontantInitial;ResteDu;JoursRetard;Tranche;ConditionPaiement;Bloque;ModeReglement;Solde;DocExterne';
                                                                               END;
                                                                                }

    { [{8CE4017D-70B8-444F-84DB-07FD045E665F}];1 ;FOURNISSEURS        ;Element ;Table   ;
                                                  SourceTable=Table23;
                                                  Export::OnPreXMLItem=BEGIN
                                                                         Vendor.SETRANGE("Date Filter",0D,DateFin);
                                                                       END;

                                                  Export::OnAfterGetRecord=BEGIN
                                                                             Vendor.CALCFIELDS("Net Change (LCY)");
                                                                             IF Vendor."Net Change (LCY)" = 0 THEN
                                                                               currXMLport.SKIP;
                                                                           END;
                                                                            }

    { [{1072A557-C878-4537-B0F4-022E3515397B}];2 ;LigneFournisseur    ;Element ;Text    ;
                                                  VariableName=LigneFournisseur;
                                                  Export::OnBeforePassVariable=BEGIN
                                                                                 // V2 : Net Change fournisseur deja positif quand COMAF doit -> plus d inversion de signe
                                                                                 // GroupeFournisseur permet de separer fournisseurs (LOC-EXP, ETR-EXP) / banques (BQ-*) / leasing (LEAS-*)
                                                                                 LigneFournisseur :=
                                                                                   'F;' + FDate(DateFin) + ';' + T(Vendor."No.") + ';' + T(Vendor.Name) + ';' +
                                                                                   T(Vendor."Purchaser Code") + ';' + T(Vendor."Vendor Posting Group") + ';;;;;;;;;' +
                                                                                   T(Vendor."Payment Terms Code") + ';' + FORMAT(Vendor.Blocked,0,2) + ';' +
                                                                                   T(Vendor."Payment Method Code") + ';' + N(Vendor."Net Change (LCY)") + ';';
                                                                               END;
                                                                                }

    { [{2A7CAF11-951B-4512-8EC3-0167D934BC07}];1 ;ECRITURES           ;Element ;Table   ;
                                                  SourceTable=Table25;
                                                  Export::OnPreXMLItem=BEGIN
                                                                         "Vendor Ledger Entry".SETRANGE("Posting Date",0D,DateFin);
                                                                         IF DateFin >= TODAY THEN
                                                                           "Vendor Ledger Entry".SETRANGE(Open,TRUE);
                                                                         "Vendor Ledger Entry".SETRANGE("Date Filter",0D,DateFin);
                                                                       END;

                                                  Export::OnAfterGetRecord=BEGIN
                                                                             "Vendor Ledger Entry".CALCFIELDS("Amount (LCY)","Remaining Amt. (LCY)");
                                                                             IF "Vendor Ledger Entry"."Remaining Amt. (LCY)" = 0 THEN
                                                                               currXMLport.SKIP;
                                                                             Ech := "Vendor Ledger Entry"."Due Date";
                                                                             IF Ech = 0D THEN
                                                                               Ech := "Vendor Ledger Entry"."Posting Date";
                                                                             Jours := Retard(Ech,DateFin);
                                                                             NomFrs("Vendor Ledger Entry"."Vendor No.");
                                                                           END;
                                                                            }

    { [{F8FD0385-F2D4-4294-BA9B-C6A045C736D7}];2 ;LigneEcriture       ;Element ;Text    ;
                                                  VariableName=LigneEcriture;
                                                  Export::OnBeforePassVariable=BEGIN
                                                                                 // ecritures : signe inverse (positif = dette de COMAF)
                                                                                 LigneEcriture :=
                                                                                   'E;' + FDate(DateFin) + ';' + T("Vendor Ledger Entry"."Vendor No.") + ';' + T(NomFournisseur) + ';' +
                                                                                   T("Vendor Ledger Entry"."Purchaser Code") + ';' + T("Vendor Ledger Entry"."Vendor Posting Group") + ';' +
                                                                                   T("Vendor Ledger Entry"."Document No.") + ';' + FORMAT("Vendor Ledger Entry"."Document Type",0,2) + ';' +
                                                                                   FDate("Vendor Ledger Entry"."Posting Date") + ';' + FDate(Ech) + ';' +
                                                                                   N(-"Vendor Ledger Entry"."Amount (LCY)") + ';' + N(-"Vendor Ledger Entry"."Remaining Amt. (LCY)") + ';' +
                                                                                   I(Jours) + ';' + Tranche(Jours) + ';;;' + T("Vendor Ledger Entry"."Payment Method Code") + ';;' +
                                                                                   T("Vendor Ledger Entry"."External Document No.");
                                                                               END;
                                                                                }

    { [{09BF9ED4-0AAC-462B-9AA7-ECB8157F6A1C}];1 ;DECAISSEMENTS       ;Element ;Table   ;
                                                  VariableName=Paiement;
                                                  SourceTable=Table25;
                                                  Export::OnPreXMLItem=BEGIN
                                                                         // V2 : tous les mouvements debiteurs hors factures et avoirs (reglements, traites, virements)
                                                                         Paiement.SETFILTER("Document Type",'<>%1&<>%2',Paiement."Document Type"::Invoice,Paiement."Document Type"::"Credit Memo");
                                                                         IF DateDebut = 0D THEN
                                                                           Paiement.SETRANGE("Entry No.",0)
                                                                         ELSE
                                                                           Paiement.SETRANGE("Posting Date",DateDebut,DateFin);
                                                                       END;

                                                  Export::OnAfterGetRecord=BEGIN
                                                                             Paiement.CALCFIELDS("Amount (LCY)");
                                                                             IF (Paiement."Amount (LCY)" <= 0) OR (ABS(Paiement."Amount (LCY)") < 1) THEN
                                                                               currXMLport.SKIP;
                                                                             NomFrs(Paiement."Vendor No.");
                                                                           END;
                                                                            }

    { [{193F5283-070D-4AB7-99CC-8B7F61A3B26D}];2 ;LignePaiement       ;Element ;Text    ;
                                                  VariableName=LignePaiement;
                                                  Export::OnBeforePassVariable=BEGIN
                                                                                 LignePaiement :=
                                                                                   'P;;' + T(Paiement."Vendor No.") + ';' + T(NomFournisseur) + ';' +
                                                                                   T(Paiement."Purchaser Code") + ';' + T(Paiement."Vendor Posting Group") + ';' +
                                                                                   T(Paiement."Document No.") + ';' + FORMAT(Paiement."Document Type",0,2) + ';' +
                                                                                   FDate(Paiement."Posting Date") + ';;' + N(Paiement."Amount (LCY)") + ';;;;;;' +
                                                                                   T(Paiement."Payment Method Code") + ';;' + T(Paiement."External Document No.");
                                                                               END;
                                                                                }

  CODE
  {
    VAR
      DateDebut@1000000100 : Date;
      DateFin@1000000101 : Date;
      Ech@1000000102 : Date;
      Jours@1000000103 : Integer;
      Vend@1000000104 : Record 23;
      NomFournisseur@1000000105 : Text[100];
      CacheFrs@1000000106 : Code[20];

    PROCEDURE FDate@1000000010(D@1000000000 : Date) : Text[10];
    BEGIN
      IF D = 0D THEN
        EXIT('');
      EXIT(FORMAT(D,0,'<Year4>-<Month,2>-<Day,2>'));
    END;

    PROCEDURE T@1000000011(Txt@1000000000 : Text[250]) : Text[250];
    BEGIN
      EXIT(DELCHR(CONVERTSTR(Txt,';',','),'=','"'));
    END;

    PROCEDURE N@1000000012(Val@1000000000 : Decimal) : Text[30];
    BEGIN
      EXIT(FORMAT(Val,0,9));
    END;

    PROCEDURE I@1000000013(Val@1000000000 : Integer) : Text[30];
    BEGIN
      EXIT(FORMAT(Val,0,9));
    END;

    PROCEDURE Tranche@1000000014(J@1000000000 : Integer) : Text[10];
    BEGIN
      IF J <= 0 THEN
        EXIT('NON ECHU');
      IF J <= 30 THEN
        EXIT('1-30');
      IF J <= 60 THEN
        EXIT('31-60');
      IF J <= 90 THEN
        EXIT('61-90');
      IF J <= 180 THEN
        EXIT('91-180');
      IF J <= 365 THEN
        EXIT('181-365');
      EXIT('+365');
    END;

    PROCEDURE Retard@1000000015(Ech@1000000000 : Date;Ref@1000000001 : Date) : Integer;
    BEGIN
      IF (Ech = 0D) OR (Ech >= Ref) THEN
        EXIT(0);
      EXIT(Ref - Ech);
    END;

    PROCEDURE NomFrs@1000000018(No@1000000000 : Code[20]);
    BEGIN
      IF No = CacheFrs THEN
        EXIT;
      CacheFrs := No;
      NomFournisseur := '';
      IF Vend.GET(No) THEN
        NomFournisseur := Vend.Name;
    END;

    BEGIN
    END.
  }
