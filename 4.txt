    VAR
      DateDebut@1000000100 : Date;
      DateFin@1000000101 : Date;
      Ech@1000000102 : Date;
      Jours@1000000103 : Integer;
      Vend@1000000104 : Record 23;
      NomFournisseur@1000000105 : Text[100];
      CacheFrs@1000000106 : Code[20];
    PROCEDURE FDate@1000000010(D@1000000000 : Date) : Text[10];
    BEGIN
      IF D = 0D THEN
        EXIT('');
      EXIT(FORMAT(D,0,'<Year4>-<Month,2>-<Day,2>'));
    END;

    PROCEDURE T@1000000011(Txt@1000000000 : Text[250]) : Text[250];
    BEGIN
      EXIT(DELCHR(CONVERTSTR(Txt,';',','),'=','"'));
    END;

    PROCEDURE N@1000000012(Val@1000000000 : Decimal) : Text[30];
    BEGIN
      EXIT(FORMAT(Val,0,9));
    END;

    PROCEDURE I@1000000013(Val@1000000000 : Integer) : Text[30];
    BEGIN
      EXIT(FORMAT(Val,0,9));
    END;

    PROCEDURE Tranche@1000000014(J@1000000000 : Integer) : Text[10];
    BEGIN
      IF J <= 0 THEN EXIT('NON ECHU');
      IF J <= 30 THEN EXIT('1-30');
      IF J <= 60 THEN EXIT('31-60');
      IF J <= 90 THEN EXIT('61-90');
      IF J <= 180 THEN EXIT('91-180');
      IF J <= 365 THEN EXIT('181-365');
      EXIT('+365');
    END;

    PROCEDURE Retard@1000000015(Ech@1000000000 : Date;Ref@1000000001 : Date) : Integer;
    BEGIN
      IF (Ech = 0D) OR (Ech >= Ref) THEN
        EXIT(0);
      EXIT(Ref - Ech);
    END;

    PROCEDURE NomFrs@1000000018(No@1000000000 : Code[20]);
    BEGIN
      IF No = CacheFrs THEN
        EXIT;
      CacheFrs := No;
      NomFournisseur := '';
      IF Vend.GET(No) THEN
        NomFournisseur := Vend.Name;
    END;

    BEGIN
    END.
