    PROCEDURE Tranche@1000000014(J@1000000000 : Integer) : Text[10];
    BEGIN
      IF J <= 0 THEN
        EXIT('NON ECHU');
      IF J <= 30 THEN
        EXIT('1-30');
      IF J <= 60 THEN
        EXIT('31-60');
      IF J <= 90 THEN
        EXIT('61-90');
      IF J <= 180 THEN
        EXIT('91-180');
      IF J <= 365 THEN
        EXIT('181-365');
      EXIT('+365');
    END;
